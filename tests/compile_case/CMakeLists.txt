cmake_minimum_required(VERSION 3.20)
project(DimenGuardCompileCase LANGUAGES CXX)
add_executable(compile_case "${DG_ROOT}/tests/compile_fail/${DG_CASE}.cpp")
target_include_directories(compile_case PRIVATE "${DG_ROOT}/include")
target_compile_features(compile_case PRIVATE cxx_std_20)
set_target_properties(compile_case PROPERTIES CXX_EXTENSIONS OFF)
if(DG_NEGATIVE)
    target_compile_definitions(compile_case PRIVATE DIMENGUARD_NEGATIVE_CASE=1)
endif()
